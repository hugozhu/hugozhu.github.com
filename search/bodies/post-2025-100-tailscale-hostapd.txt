将接入tailnet的树莓派hostapd client通过tailscale exit节点上网
配置如下 sudo nohup ./tailscaled &gt; /dev/null 2&gt;&amp;1 &amp; sudo ./tailscale up sudo ./tailscale set --exit-node=&lt;your_node&gt; --exit-node-allow-lan-access sudo iptables -t nat -A POSTROUTING -o tailscale0 -j MASQUERADE sudo iptables -A FORWARD -i wlan0 -o tailscale0 -m state --state RELATED,ESTABLISHED -j ACCEPT sudo iptables -A FORWARD -i tailscale0 -o wlan0 -j ACCEPT
