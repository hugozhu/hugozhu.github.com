self.write4bits(self.LCD_DISPLAYCONTROL | self.displaycontrol) def noBlink(self): &#34;&#34;&#34; Turn on and off the blinking cursor &#34;&#34;&#34; self.displaycontrol &amp;= ~self.LCD_BLINKON self.write4bits(self.LCD_DISPLAYCONTROL | self.displaycontrol) def noBlink(self): &#34;&#34;&#34; Turn on and off the blinking cursor &#34;&#34;&#34; self.displaycontrol &amp;= ~self.LCD_BLINKON &#34;&#34;&#34; Send command to LCD &#34;&#34;&#34; self.delayMicroseconds(1000) # 1000 microsecond sleep bits=bin(bits)[2:].zfill(8) self.GPIO.output(self.pin_rs, char_mode) for pin in self.pins_db: in range(4,8): if bits[i] == &#34;1&#34;: self.GPIO.output(self.pins_db[::-1][i-4], True) self.pulseEnable() def delayMicroseconds(self, microseconds): seconds = microseconds / float(1000000)	# divide self.GPIO.output(self.pin_e, True) self.delayMicroseconds(1)	# 1 microsecond pause - enable pulse must be &gt; 450ns self.GPIO.output(self.pin_e, False) self.delayMicroseconds(1)	# commands need &gt;
