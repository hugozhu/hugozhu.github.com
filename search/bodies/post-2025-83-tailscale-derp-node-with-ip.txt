虽然tailscale用了很多NAT穿透 NAT traversal技术，但仍然会有不能P2P自连的情况，这时候tailscale就会使用最近的derp节点来建立连接，自建derper可用于加速tailscale网络的连通性能
实现步骤 用docker-compose来编排和运行derper服务 docker-compose.yaml： version: &#34;3&#34; services: tailscale: image: tailscale/tailscale container_name: tailscale privileged: true restart: always volumes: - &#34;./tailscale/data:/var/lib/tailscale&#34; - &#34;./tailscale/tmp:/tmp&#34; - &#34;/dev/net/tun:/dev/net/tun&#34; cap_add: - net_admin - sys_module environment: TS_AUTHKEY: &#34;&lt;your_tailscale_auth_key&gt;&#34; TS_STATE_DIR: &#34;/var/lib/tailscale&#34; TS_USERSPACE: &#34;false&#34; networks: - tailscale_network dns: - 8.8.8.8 - 8.8.4.4 derper: #image: starudream/derper build: context: ./build dockerfile: Dockerfile container_name: derper restart: always #command: /tailscale/derper -a :80 -verify-clients depends_on: - tailscale ports: - &#34;3478:3478/udp&#34; - &#34;80:80/tcp&#34; - &#34;443:443/tcp&#34; volumes: - &#34;./tailscale/tmp:/var/run/tailscale&#34; networks: - tailscale_network dns: - 8.8.8.8 - 8.8.4.4 networks: tailscale_network: external: true build/Dockerfile # 编译 FROM golang:alpine AS builder # 切换模块源为中国Go模块代理服务器 RUN go env -w GOPROXY=https://goproxy.cn,direct # 拉取代码 RUN go install tailscale.com/cmd/derper@latest # 去除域名验证（删除cmd/derper/cert.go文件的91~93行） RUN find /go/pkg/mod/tailscale.com@*/cmd/derper/cert.go -type f -exec sed -i &#39;97,99d&#39; {} + # 编译 RUN derper_dir=$(find /go/pkg/mod/tailscale.com@*/cmd/derper -type d) &amp;&amp; \ cd $derper_dir &amp;&amp; \ go build -o /etc/derp/derper # 生成最终镜像 FROM alpine:latest WORKDIR /apps COPY --from=builder /etc/derp/derper . RUN ln -sf /usr/share/zoneinfo/Asia/Shanghai /etc/localtime \ &amp;&amp; echo &#39;Asia/Shanghai&#39; &gt; /etc/timezone ENV LANG C.UTF-8 # 创建软链接 解决二进制无法执行问题 Amd架构必须执行，Arm不需要执行 RUN mkdir /lib64 &amp;&amp; ln -s /lib/libc.musl-x86_64.so.1 /lib64/ld-linux-x86-64.so.2 # 添加源 RUN sed -i &#39;s/dl-cdn.alpinelinux.org/mirrors.tuna.tsinghua.edu.cn/g&#39; /etc/apk/repositories # 安装openssl RUN apk add openssl &amp;&amp; mkdir /ssl ENV HOST_NAME=&lt;your_domain&gt; RUN echo &#34;Building derper for $HOST_NAME&#34; RUN openssl req -x509 -newkey rsa:4096 -sha256 -days 3650 -nodes -keyout /ssl/$HOST_NAME.key -out /ssl/$HOST_NAME.crt -subj &#34;/CN=$HOST_NAME&#34; -addext &#34;subjectAltName=DNS:$HOST_NAME&#34; CMD ./derper -hostname $HOST_NAME -certmode manual -certdir /ssl --verify-clients=true 验证 用 curl -k https://&lt;ip&gt;/ 验证Derp服务正常 用 echo &quot;Hello, Server&quot; | nc -u &lt;host&gt; 3478 验证UTUN端口正常，nc -u -l 3478 可以打开一个仿真的UDP Server来验证防火墙是否打开 添加新增derp节点到Tailscale Open https://login.tailscale.com/admin/acls/file
